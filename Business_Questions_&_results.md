## 1.5 Books Generating High Revenue Despite Low Sales Volume

**Business Question:** Which books generated high revenue despite having relatively low sales volume?

```sql
WITH book_sales_summary AS (
    SELECT
        b.Title,
        SUM(o.Quantity) AS copies_sold,
        SUM(o.Total_Amount) AS revenue_generated
    FROM Orders o
    JOIN Books b
        ON b.Book_ID = o.Book_ID
    GROUP BY b.Title
)

SELECT
    Title,
    copies_sold,
    revenue_generated
FROM book_sales_summary
WHERE
    copies_sold < (SELECT AVG(copies_sold) FROM book_sales_summary)
    AND revenue_generated > (SELECT AVG(revenue_generated) FROM book_sales_summary);
```

## 2.5 Customers Purchasing More Books than the Average Customer

**Business Question:** Which customers have purchased more books than the average customer?

```sql
WITH customer_book_totals AS (
    SELECT
        c.Customer_ID,
        c.Name AS customer_name,
        SUM(o.Quantity) AS total_books_purchased
    FROM Customers c
    JOIN Orders o
        ON c.Customer_ID = o.Customer_ID
    GROUP BY
        c.Customer_ID,
        c.Name
)

SELECT
    Customer_ID,
    customer_name,
    total_books_purchased
FROM customer_book_totals
WHERE total_books_purchased > (
    SELECT AVG(total_books_purchased)
    FROM customer_book_totals
)
ORDER BY total_books_purchased DESC;
```

## 3.5 High-Selling Books That Should Be Restocked

**Business Question:** Which books have high sales but currently have low inventory, indicating they should be restocked?

```sql
WITH book_sales AS (
    SELECT
        b.Book_ID,
        b.Title,
        b.Stock,
        SUM(o.Quantity) AS total_books_sold
    FROM Books b
    JOIN Orders o
        ON b.Book_ID = o.Book_ID
    GROUP BY
        b.Book_ID,
        b.Title,
        b.Stock
)

SELECT
    Title,
    total_books_sold,
    Stock
FROM book_sales
WHERE total_books_sold > (
    SELECT AVG(total_books_sold)
    FROM book_sales
)
AND Stock < (
    SELECT AVG(Stock)
    FROM book_sales
)
ORDER BY total_books_sold DESC;
```
